import h5py, zlib, numpy as np, zstandard as zs, sys, collections
sys.stdout.reconfigure(encoding='utf-8')
CH=32768
def run(p):
    f=h5py.File(p,'r')
    ds=[]
    def v(n,o):
        if isinstance(o,h5py.Dataset) and o.dtype==np.float64 and n.startswith('results/') and '/solution_' in n: ds.append((n,o))
    f.visititems(v)
    kinds=collections.defaultdict(lambda:[0,0,0,0])  # raw, deflate3, zstd3, ndatasets
    dup=[]
    for n,o in ds:
        kind='cell_fields' if '/cell_fields/' in n else 'fields'
        a=o[...].ravel()
        if kind=='cell_fields' and len(dup)<3 and a.size>1000:
            dup.append((n,a.size,np.unique(a).size))
        for i in range(0,a.size,CH):
            c=np.ascontiguousarray(a[i:i+CH]).tobytes()
            k=kinds[kind]; k[0]+=len(c); k[1]+=len(zlib.compress(c,3)); k[2]+=len(zs.ZstdCompressor(level=3).compress(c)); 
        kinds[kind][3]+=1
    nm=f['mesh/points'].shape[0]; nc=f['mesh/cells'].shape[0]
    print(p.split('/')[-4] if 'nyx_runs' in p else p, 'size', round(__import__('os').path.getsize(p)/1e6,1),'MB; mesh pts',nm,'cells',nc)
    for k,(r,d,z,n) in kinds.items():
        print(f'  {k:12s} n={n:4d} raw {r/1e6:7.1f}  deflate3 {d/1e6:7.1f} ({r/d:.2f}x)  zstd3 {z/1e6:7.1f} ({r/z:.2f}x)')
    for n,s,u in dup: print('  unique check',n.split('solution_')[1],s,'values',u,'unique')
    try:
        print('  physics',list(f['results']), [dict(f['results'][k].attrs).get('study') for k in f['results']])
    except Exception as e: print(e)
for p in sys.argv[1:]: run(p)
